pragma solidity ^0.8.26;

import {IERC20} from "@openzeppelin/contracts/token/ERC20/IERC20.sol";
import {
    SafeERC20
} from "@openzeppelin/contracts/token/ERC20/utils/SafeERC20.sol";
import {IOrderBook} from "../interfaces/IOrderBook.sol";
import {IMatchingEngine} from "../interfaces/IMatchingEngine.sol";
import {IConditionalTokens} from "../interfaces/IConditionalTokens.sol";
import {Types} from "../libraries/Types.sol";
import {Errors} from "../libraries/Errors.sol";
import {
    ReentrancyGuard
} from "@openzeppelin/contracts/utils/ReentrancyGuard.sol";

/// @title Market — 二元预测市场代理合约（EIP-1167 克隆）
/// @author ELFLAB
/// @notice 每个市场为 EIP-1167 最小代理克隆实例，作为用户与底层组件（OrderBook、MatchingEngine）之间的代理层。
///         Market 持有用户抵押的 ERC-20 collateral，托管 CTF token 仓位，并负责撮合后的：
///         1. 向 Taker mint CTF token
///         2. 向各 Maker 分配 collateral
/// @dev 不可变参数在构造时设置，之后不可更改。所有函数均使用 nonReentrant 防重入。
///      CTF mint/burn 需要 Market 先被 ConditionalTokens 授权（setAuthorized）。
contract Market is ReentrancyGuard {
    using SafeERC20 for IERC20;

    // ===== 不可变参数（构造后不可更改）=====

    /// @notice 市场使用的抵押 ERC-20 代币（如 USDC）
    IERC20 public collateral;

    /// @notice Gnosis Conditional Tokens 条件 ID
    bytes32 public conditionId;

    /// @notice 市场专属手续费率（1e6 = 100%）
    uint256 public fee;

    /// @notice 订单簿合约地址
    IOrderBook public orderBook;

    /// @notice 撮合引擎合约地址
    IMatchingEngine public matchingEngine;

    /// @notice ConditionalTokens 合约地址（用于 mint/burn CTF token）
    IConditionalTokens public conditionalTokens;

    /// @notice 创建此市场的工厂合约地址
    /// @dev EIP-1167 克隆部署时 msg.sender = factory
    address public factory;

    /// @notice 市场开始时间（Unix 时间戳），低于此时间禁止交易
    uint64 public startTime;

    /// @notice 市场结束时间（Unix 时间戳），超过此时间禁止交易
    uint64 public endTime;

    // ===== 构造函数 =====

    /// @notice 初始化市场参数（EIP-1167 克隆后调用，替代构造函数）
    /// @param _collateral        抵押 ERC-20 代币地址
    /// @param _conditionId       Conditional Tokens 条件 ID
    /// @param _fee               手续费率（1e6 = 100%）
    /// @param _startTime         市场开始时间（Unix 时间戳）
    /// @param _endTime           市场结束时间（Unix 时间戳）
    /// @param _orderBook         订单簿合约地址
    /// @param _matchingEngine    撮合引擎合约地址
    /// @param _conditionalTokens ConditionalTokens 合约地址
    /// @dev 在 EIP-1167 克隆部署后调用，每个克隆实例仅可初始化一次
    function initialize(
        IERC20 _collateral,
        bytes32 _conditionId,
        uint256 _fee,
        uint64 _startTime,
        uint64 _endTime,
        IOrderBook _orderBook,
        IMatchingEngine _matchingEngine,
        IConditionalTokens _conditionalTokens
    ) external {
        require(startTime == 0, "M: already initialized");
        require(_startTime < _endTime, "M: invalid time range");

        Errors._checkZeroAddress(address(_collateral));
        Errors._checkZeroAddress(address(_orderBook));
        Errors._checkZeroAddress(address(_matchingEngine));
        Errors._checkZeroAddress(address(_conditionalTokens));

        collateral = _collateral;
        conditionId = _conditionId;
        fee = _fee;
        startTime = _startTime;
        endTime = _endTime;
        orderBook = _orderBook;
        matchingEngine = _matchingEngine;
        conditionalTokens = _conditionalTokens;
        factory = msg.sender;
    }

    /// @notice 构造函数（EIP-1167 克隆模板需要空构造器）
    constructor() {}

    /// @notice 检查市场是否已初始化（构造后始终为 true）
    function initialized() external pure returns (bool) {
        return true;
    }

    // ===== 订单操作 =====

    /// @notice 下单（Maker）：将 collateral 转入 Market 锁定，并授权 OrderBook 支配
    /// @param isYes  true = 购买 YES 仓位，false = 购买 NO 仓位
    /// @param price  限价价格（1e18 精度）
    /// @param amount 购买数量（CTF 仓位数量）
    /// @return orderId 生成的订单 ID（由 OrderBook 分配）
    /// @dev 流程：
    ///      1. 计算 cost = price * amount
    ///      2. 从用户转入 cost 数量的 collateral 到 Market（锁定）
    ///      3. 增加 Market 对 OrderBook 的 allowance
    ///      4. 调用 OrderBook.placeOrder() 实际创建订单
    function placeOrder(
        bool isYes,
        uint128 price,
        uint128 amount
    ) external nonReentrant returns (uint64 orderId) {
        Errors._checkZeroAmount(amount);
        require(price > 0 && price <= 1e18, "M: invalid price");
        require(
            block.timestamp >= startTime && block.timestamp <= endTime,
            "M: market not active"
        );

        uint256 cost = (uint256(price) * amount) / 1e18;
        collateral.safeTransferFrom(msg.sender, address(this), cost);
        _ensureAllowance(address(orderBook), cost);

        orderId = orderBook.placeOrder(
            uint64(uint256(conditionId)),
            isYes,
            price,
            amount
        );
        emit OrderPlaced(msg.sender, isYes, price, amount);
    }

    /// @notice 取消订单（Maker）：验证订单归属后，退还剩余未成交 collateral
    /// @param orderId 要取消的订单 ID
    /// @dev 退还金额 = (amount - filled) * price
    function cancelOrder(uint64 orderId) external nonReentrant {
        Types.OrderData memory o = orderBook.getOrder(orderId);
        require(msg.sender == o.maker, "M: not your order");
        require(o.isActive && o.filled < o.amount, "M: cannot cancel");

        uint128 unfilled = o.amount - o.filled;
        uint256 refund = (uint256(unfilled) * o.price) / 1e18;

        collateral.safeTransfer(msg.sender, refund);
        orderBook.cancelOrder(orderId);
        emit OrderCancelled(orderId);
    }

    /// @notice 执行订单（Taker）：撮合核心
    /// @param isYes       Taker 方向：true = 买入 YES，false = 买入 NO
    /// @param limitPrice  Taker 限价（价格上限）
    /// @param amount      期望成交数量
    /// @param minFill     最小成交数量下限
    /// @return filled   实际成交数量
    /// @return avgPrice 成交均价（加权平均）
    /// @dev 完整撮合流程：
    ///      1. Taker 预付 cost = limitPrice * amount（可能多付）
    ///      2. 调用 MatchingEngine.matchOrder() → OrderBook.executeOrderWithMinFill()
    ///         获取 fills[]（每笔成交含 maker/amount/price）
    ///      3. 计算加权均价 avgPrice 和实际总成本
    ///      4. 向 Taker mint CTF token（outcomeIndex = isYes ? 0 : 1）
    ///      5. 向各 Maker 分配 collateral（makerFillAmount * makerPrice）
    ///      6. 退还 Taker 多付部分（预付 - 实际成本 - taker手续费已在ME中扣）
    /// !note Market 须已被 ConditionalTokens 授权（setAuthorized）才能 mint
    function fillOrder(
        bool isYes,
        uint128 limitPrice,
        uint128 amount,
        uint128 minFill
    ) external nonReentrant returns (uint128 filled, uint128 avgPrice) {
        Errors._checkZeroAmount(amount);
        require(
            block.timestamp >= startTime && block.timestamp <= endTime,
            "M: market not active"
        );

        // 1. Taker 预付（按 limitPrice 上限）
        uint256 takerPrepay = uint256(limitPrice) * amount;
        collateral.safeTransferFrom(msg.sender, address(this), takerPrepay);

        // 2. 调用撮合引擎获取成交明细
        (IOrderBook.Fill[] memory fills, uint128 totalFilled) = matchingEngine
            .matchOrder(
                uint64(uint256(conditionId)),
                isYes,
                limitPrice,
                amount
            );

        require(totalFilled >= minFill, "M: below min fill");

        // 3. 计算加权均价（按各 Maker 成交价加权）
        {
            uint256 weightedPriceSum = 0;
            for (uint256 i = 0; i < fills.length; i++) {
                weightedPriceSum += uint256(fills[i].price) * fills[i].amount;
            }
            avgPrice = totalFilled > 0
                ? uint128(weightedPriceSum / totalFilled)
                : 0;
        }

        // 4. 计算 Taker 实际成本（加权均价 × 成交数量）
        // uint256 takerCost = uint256(avgPrice) * totalFilled;

        // 5. 向 Taker mint CTF token（outcomeIndex = isYes ? 0 : 1）
        {
            uint256 outcomeIndex = isYes ? 0 : 1;
            conditionalTokens.mintPosition(
                msg.sender,
                conditionId,
                outcomeIndex,
                totalFilled
            );
        }

        // 6. 向各 Maker 分配 collateral（从 Market 余额转出）
        // 按 maker 地址合并多笔付款，只转一次
        // 用固定-size 数组模拟累计：paid[i]=maker, paidAmount[i]=该 maker 累计总额
        // fills.length <= 100，O(n^2) 可接受
        address[] memory paid = new address[](fills.length);
        uint256[] memory paidAmount = new uint256[](fills.length);
        uint256 paidCount;
        for (uint256 i = 0; i < fills.length; i++) {
            address maker = fills[i].maker;
            uint256 payment = uint256(fills[i].amount) * fills[i].price;
            if (payment == 0) continue;
            // 累加到 paid 数组
            bool found = false;
            for (uint256 j = 0; j < paidCount; j++) {
                if (paid[j] == maker) {
                    paidAmount[j] += payment;
                    found = true;
                    break;
                }
            }
            if (!found) {
                paid[paidCount] = maker;
                paidAmount[paidCount] = payment;
                paidCount++;
            }
        }
        // 统一转出
        for (uint256 i = 0; i < paidCount; i++) {
            if (paidAmount[i] > 0) {
                collateral.safeTransfer(paid[i], paidAmount[i]);
            }
        }

        // 7. 退还 Taker 多付部分（预付 - 实际成本）
        if (takerPrepay > uint256(avgPrice) * totalFilled) {
            collateral.safeTransfer(
                msg.sender,
                takerPrepay - uint256(avgPrice) * totalFilled
            );
        }

        emit OrderFilled(msg.sender, isYes, totalFilled, avgPrice);
    }

    /// @notice 通知型函数：用户自行在 ConditionalTokens 合约赎回仓位
    /// @param user 持仓用户地址（仅用于事件记录，无操作）
    /// @dev 此合约不托管用户仓位，赎回由用户在 ConditionalTokens 合约直接调用
    ///      conditionalTokens.redeemPositions(...) 完成。此函数仅发出事件供索引用。
    function redeem(address user) external {
        emit Redeemed(user, conditionId);
    }

    // ===== 内部函数 =====

    /// @notice 确保 Market 对目标合约有足够的 allowance
    /// @param target 目标合约地址
    /// @param amount 需要的 allowance 数量
    function _ensureAllowance(address target, uint256 amount) internal {
        if (IERC20(collateral).allowance(address(this), target) < amount) {
            if (IERC20(collateral).allowance(address(this), target) > 0) {
                collateral.safeDecreaseAllowance(target, type(uint256).max);
            }
            collateral.safeIncreaseAllowance(target, amount);
        }
    }

    // ===== 事件 =====

    event OrderPlaced(
        address indexed maker,
        bool isYes,
        uint128 price,
        uint128 amount
    );
    event OrderCancelled(uint64 indexed orderId);
    event OrderFilled(
        address indexed taker,
        bool isYes,
        uint128 filled,
        uint128 avgPrice
    );
    event Redeemed(address indexed user, bytes32 indexed conditionId);
}
