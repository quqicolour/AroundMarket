pragma solidity ^0.8.26;

import {Ownable} from "@openzeppelin/contracts/access/Ownable.sol";
import {IERC20} from "@openzeppelin/contracts/token/ERC20/IERC20.sol";
import {Clones} from "@openzeppelin/contracts/proxy/Clones.sol";
import {Types} from "../libraries/Types.sol";
import {Errors} from "../libraries/Errors.sol";
import {Market} from "./Market.sol";
import {OrderBook} from "./OrderBook.sol";
import {MatchingEngine} from "./MatchingEngine.sol";
import {IOrderBook} from "../interfaces/IOrderBook.sol";
import {IMatchingEngine} from "../interfaces/IMatchingEngine.sol";
import {ISettlementManager} from "../interfaces/ISettlementManager.sol";
import {IConditionalTokens} from "../interfaces/IConditionalTokens.sol";

/// @title PredictionMarketFactory — 预测市场工厂
/// @author ELFLAB
/// @notice 使用 EIP-1167 克隆部署 OrderBook/MatchingEngine（无状态），Market 采用 CREATE3
///         模式通过最小代理克隆（EIP-1167）并调用 initialize() 初始化。
///         工厂是市场的中心注册表，记录所有市场的元数据（collateral、orderBook、matchingEngine 等）。
///         平台收取的手续费通过 feeRecipient 和 platformFee（默认 0.02%）配置。
contract PredictionMarketFactory is Ownable {
    /// @notice OrderBook 模板合约地址（无状态，可被多个市场克隆共享）
    address public orderBookTemplate;

    /// @notice MatchingEngine 模板合约地址（无状态，可被多个市场克隆共享）
    address public matchingEngineTemplate;

    /// @notice Market 模板合约地址（部署时需先初始化模板）
    address public marketTemplate;

    /// @notice 平台手续费接收地址
    /// @dev 市场单独设置 fee 时优先使用市场自身费率，否则使用此全局平台费率
    address public feeRecipient;

    /// @notice 全局平台手续费率（单位：1e6 = 100%）
    /// @dev 默认 200 = 0.02%。市场创建时若 _fee 传 0，则自动使用此值
    uint32 public platformFee = 200;

    uint32 public luckyFee = 100;

    /// @notice 下一个可用的市场 ID（自增，单调递增，不重复）
    uint64 public nextMarketId = 1;

    /// @notice 市场 ID → 市场元数据
    /// @dev 包含 collateral、orderBook、matchingEngine、conditionId、resolved、fee
    mapping(uint64 => Types.MarketData) public markets;

    /// @notice 市场 ID → 是否已创建（用于 existence 检查）
    mapping(uint64 => bool) public marketExists;

    /// @notice SettlementManager 合约地址（用于自动注册市场）
    ISettlementManager public settlementManager;

    /// @notice ConditionalTokens 合约地址（用于 Market 克隆时传入）
    IConditionalTokens public conditionalTokens;

    // ===== 事件 =====

    /// @notice 创建新市场时触发
    /// @param marketId    新市场 ID（全局唯一）
    /// @param market      直接 new 部署的 Market 合约地址
    /// @param collateral  市场使用的抵押代币地址
    /// @param conditionId Gnosis Conditional Tokens 的条件 ID（由工厂外部计算后传入）
    event MarketCreated(
        uint64 indexed marketId,
        address indexed market,
        address collateral,
        bytes32 conditionId
    );

    /// @notice 更新手续费接收地址时触发
    /// @param newRecipient 新的手续费接收地址
    event FeeRecipientUpdated(address indexed newRecipient);

    /// @notice 更新平台手续费率时触发
    /// @param newFee 新的平台手续费率（1e6 = 100%）
    event PlatformFeeUpdated(uint32 newFee);

    /// @notice 更新幸运奖池手续费率时触发
    /// @param newFee 新的幸运奖池手续费率（1e6 = 100%）
    event LuckyFeeUpdated(uint32 newFee);

    /// @notice 更新模板合约地址时触发
    /// @param orderBook       新的 OrderBook 模板地址
    /// @param matchingEngine  新的 MatchingEngine 模板地址
    /// @param market          新的 Market 模板地址
    event TemplatesUpdated(
        address indexed orderBook,
        address indexed matchingEngine,
        address indexed market
    );

    // ===== 构造函数 =====

    /// @notice 初始化工厂所有者
    /// @param _owner 工厂 Owner（Ownable）
    constructor(address _owner) Ownable(_owner) {}

    // ===== 配置函数（仅 Owner） =====

    /// @notice 设置三个模板合约地址，模板必须先部署并（对于 Market 模板）完成初始化
    /// @param _orderBookTemplate       OrderBook 模板地址
    /// @param _matchingEngineTemplate MatchingEngine 模板地址
    /// @param _marketTemplate          Market 模板地址（应在工厂部署后先由工厂初始化）
    /// @dev Market 模板需要持有正确的 orderBook/matchingEngine 地址，因此建议先设置
    ///      orderBookTemplate 和 matchingEngineTemplate，再部署 Market 模板，最后调用此函数
    function setTemplates(
        address _orderBookTemplate,
        address _matchingEngineTemplate,
        address _marketTemplate
    ) external onlyOwner {
        Errors._checkZeroAddress(_orderBookTemplate);
        Errors._checkZeroAddress(_matchingEngineTemplate);
        Errors._checkZeroAddress(_marketTemplate);
        orderBookTemplate = _orderBookTemplate;
        matchingEngineTemplate = _matchingEngineTemplate;
        marketTemplate = _marketTemplate;
        emit TemplatesUpdated(
            _orderBookTemplate,
            _matchingEngineTemplate,
            _marketTemplate
        );
    }

    /// @notice 设置平台手续费接收地址
    /// @param _feeRecipient 接收手续费的 EOA 或合约地址
    function setFeeRecipient(address _feeRecipient) external onlyOwner {
        Errors._checkZeroAddress(_feeRecipient);
        feeRecipient = _feeRecipient;
        emit FeeRecipientUpdated(_feeRecipient);
    }

    /// @notice 设置全局平台手续费率
    /// @param _platformFee 手续费率（0 ~ 1e6，1e6 = 100%）
    /// @dev 市场单独指定 fee 时覆盖此值；传 0 则表示使用平台默认费率
    function setPlatformFee(uint32 _platformFee) external onlyOwner {
        require(_platformFee <= 1e6, "Factory: fee too high");
        platformFee = _platformFee;
        emit PlatformFeeUpdated(_platformFee);
    }

    /// @notice 设置全局幸运手续费率
    /// @param _luckyFee 幸运手续费率（0 ~ 1e6，1e6 = 100%）
    /// @dev 市场单独指定 fee 时覆盖此值；传 0 则表示使用平台默认费率
    function setLuckyFee(uint32 _luckyFee) external onlyOwner {
        require(_luckyFee <= 1e6, "Factory: fee too high");
        luckyFee = _luckyFee;
        emit LuckyFeeUpdated(_luckyFee);
    }

    /// @notice 设置 SettlementManager 合约地址（用于 createMarket 时自动注册）
    /// @param _settlementManager SettlementManager 合约地址
    /// @dev 需将 Factory owner 设为 SM，这样 createMarket 才能调用 SM.registerMarket
    function setSettlementManager(
        address _settlementManager
    ) external onlyOwner {
        Errors._checkZeroAddress(_settlementManager);
        settlementManager = ISettlementManager(_settlementManager);
    }

    /// @notice 设置 ConditionalTokens 合约地址（Market 克隆时需传入此地址）
    /// @param _conditionalTokens ConditionalTokens 合约地址
    function setConditionalTokens(
        address _conditionalTokens
    ) external onlyOwner {
        Errors._checkZeroAddress(_conditionalTokens);
        conditionalTokens = IConditionalTokens(_conditionalTokens);
    }

    // ===== 市场创建 =====

    /// @notice 创建新的二元预测市场（EIP-1167 克隆模式）
    /// @param collateral   抵押 ERC-20 代币地址（如 USDC）
    /// @param conditionId  Gnosis Conditional Tokens 条件 ID（由外部根据 oracle/questionId 计算）
    /// @param _fee         市场专属手续费率（传 0 则使用 platformFee 全局值）
    /// @param _startTime   市场开始时间（Unix 时间戳），低于此时间禁止交易
    /// @param _endTime     市场结束时间（Unix 时间戳），超过此时间禁止交易
    /// @return marketId 新市场的唯一 ID
    /// @dev 部署策略：
    ///      - OrderBook/MatchingEngine：EIP-1167 克隆（无状态模板，~45 字节/每个）
    ///      - Market：EIP-1167 最小代理克隆 → 调用 initialize()（避免构造参数编码进 code blob）
    ///      salt = keccak256(abi.encode(marketId)) 保证市场地址可预测
    function createMarket(
        address collateral,
        bytes32 conditionId,
        uint32 _fee,
        uint64 _startTime,
        uint64 _endTime
    ) external onlyOwner returns (uint64 marketId) {
        if (collateral == address(0) || feeRecipient == address(0))
            revert Errors.ZeroAddress();
        if (orderBookTemplate == address(0) || marketTemplate == address(0))
            revert Errors.ZeroAddress();
        require(_startTime < _endTime, "Factory: invalid time range");

        marketId = nextMarketId++;

        // 1. EIP-1167 克隆 OrderBook（无状态）
        address orderBook = Clones.clone(orderBookTemplate);

        // 2. EIP-1167 克隆 MatchingEngine（无状态）
        address matchingEngine = Clones.clone(matchingEngineTemplate);

        // 3. EIP-1167 最小代理克隆 Market，并调用 initialize()
        uint32 actualFee = _fee > 0 ? _fee : platformFee;
        bytes32 salt = keccak256(abi.encode(marketId));
        address market = Clones.cloneDeterministic(marketTemplate, salt);
        Market(market).initialize(
            IERC20(collateral),
            conditionId,
            actualFee,
            _startTime,
            _endTime,
            IOrderBook(orderBook),
            IMatchingEngine(matchingEngine),
            conditionalTokens
        );

        // 4. 持久化市场元数据
        markets[marketId] = Types.MarketData({
            creator: msg.sender,
            market: market,
            collateral: collateral,
            conditionTokens: address(0),
            orderBook: orderBook,
            matchingEngine: matchingEngine,
            conditionId: conditionId,
            startTime: _startTime,
            endTime: _endTime,
            resolved: false,
            fee: actualFee
        });

        marketExists[marketId] = true;
        emit MarketCreated(marketId, market, collateral, conditionId);

        // 5. 自动注册到 SettlementManager（Factory 需为 SM 的 owner）
        if (address(settlementManager) != address(0)) {
            settlementManager.registerMarket(marketId, market);
        }
    }

    // ===== 查询函数 =====

    /// @notice 根据市场 ID 查询市场完整元数据
    /// @param marketId 市场 ID
    /// @return Types.MarketData 市场元数据结构体
    /// 如果市场不存在则 revert
    function getMarket(
        uint64 marketId
    ) external view returns (Types.MarketData memory) {
        require(marketExists[marketId], "Factory: market not exists");
        return markets[marketId];
    }

    /// @notice 获取已创建市场的总数
    /// @return nextMarketId - 1（即最后一个分配的市场 ID）
    function getMarketCount() external view returns (uint64) {
        return nextMarketId - 1;
    }
}
